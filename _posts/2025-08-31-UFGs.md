<script type="text/javascript">
(function() {
    function init() {
        try {
            // CHART DRAWING LOGIC
            var link_prefix = "https://en.wikipedia.org/wiki/";
            var tree_text = {name: "Input Indicators for Systematic Assessment of Urban Food Garden", children:[
              {name: "Quantitative \nSustainability", children:[
                {name: "Resource Inputs (Materials & Energy)", children:[{name: "Infrastructure Materials"},{name: "Substrate & Grow Medias"},{name: "Fertilizers & Soil Amendmentst"},{name: "Water (tap water, irrigation)"},{name: "Energy (electricity, fuel, diesel)"},{name: "Pesticides & Herbicides"},{name: "Packaging Materials"},{name: "Other Inputs (seeds, earthworms)"},]},
                {name: "Economic Inputs", children:[{name: "Operational Costs", article: "Operating_expense"},{name: "Capital Costs", article: "Capital_expenditure"},{name: "Land Rent / Value", article: "Land_value_tax"}]},
                {name: "System & Production Parameters", children:[{name: "Crop Type & Yield Data"},{name: "Life Cycle Stages"},{name: "Food Waste Ratios"},]},
              ]},
              {name: "Spatial Planning \n& Optimization", children:[
                {name: "Spatial & Geographic Data", children:[{name: "Land Use / Land Cover", article: "Land_use,_land-use_change,_and_forestry"},{name: "Building Data"},{name: "Infrastructure Networks"},{name: "Terrain data"},{name: "Climate Data"},{name: "Proximity to Features"},{name: "Soil Properties"},]},
                {name: "Socio-Economic & Demographic Data", children:[{name: "Population Data", article: "Demography"},{name: "Economic Indicators"},{name: "Health & Vulnerability Indicators "},{name: "Social Indicators"},]},
                {name: "Policy & Planning Scenarios", children:[{name: "Land-Use Scenarios"},{name: "Siting Criteria"},{name: "Development Goals & Constraints"},{name: "Budget / Resource Constraints"},]},
              ]},
              {name: "Systems & Nexus \nModeling", children:[
                {name: "Resource Flow & Consumption Data", children:[{name: "Water Demand & Supply Data"},{name: "Energy Demand & Supply Data"},{name: "Food Demand & Supply Data"},{name: "Organic Waste Generation Rates"},]},
                {name: "System Design & Operational Parameters", children:[{name: "Technology Specifications"},{name: "UA Parameters"},{name: "Agent Attributes"},]},
                {name: "External Drivers & Scenarios", children:[{name: "Climate Change Projections", article: "Climate_model"},{name: "Policy Scenarios"},{name: "Economic Scenarios (price inflation)"},]},
                {name: "Biophysical & Ecological Models", children:[{name: "Plant Growth Models"},{name: "Evapotranspiration Models"},{name: "Hydrological Models"},]},
              ]},
              {name: "Socio-Ecological \n& Community", children:[
                {name: "Participant & Community Demographics", children:[{name: "Socio-Economic Data"},{name: "Gardener Experience & Skill Level"},]},
                {name: "Social & Institutional Factors", children:[{name: "Participant Motivations"},{name: "Stakeholder Perceptions"},{name: "Governance Structures"},{name: "Social Networks & Cohesion"},]},
                {name: "Livelihood Capitals", children:[{name: "Human Capital", article: "Human_capital"},{name: "Social Capital", article: "Social_capital"},{name: "Financial Capital", article: "Financial_capital"}]},
                {name: "UA Site Characteristics", children:[{name: "Food Distribution Model"},{name: "Management Practices"},{name: "Site Typology"},]},
              ]},
              {name: "Empirical \n& Field-Based", children:[
                {name: "Resource Inputs (Measured)", children:[{name: "Irrigation Water Volume"},{name: "Soil Amendments"},{name: "Fuel for Machinery"},]},
                {name: "Biophysical Site Characteristics", children:[{name: "Site & Cultivated Area (m²)"},{name: "Soil Quality Metrics"},{name: "Floral & Vegetative Diversity"},]},
                {name: "Human Activity", children:[{name: "Labor Hours"},{name: "Volunteer time"},]},
              ]},
            ]};
            var tree_colors = ["#098","#07A","#490","#FA8072","#9932CC"];
            var tree_radii = [142,284,428];
            var title_text_size = 32;
            var category_text_size = 32;
            var subcategory_text_size = 15;
            var entry_text_size = 11;
            var title_y = -850;
            var sub_category_spacing = 1.0;
            var category_spacing = 0.0;
            var small_dot_radius = 3.0;
            var big_dot_radius = 10.0;
            var circumcircle_radius = 630.0;
            var circumcircle_color = "#CCCCCC";
            var polar_to_cartesian = function(radius, angle, full_angle){
              var angle_in_radians = angle * 2*Math.PI / full_angle;
              var x = radius * Math.sin(angle_in_radians);
              var y =-radius * Math.cos(angle_in_radians);
              return {x:x,y:y};
            };
            var lerp = function(a, b, t){ return (1.0-t)*a + t*b; };
            var push_radial_path = function(string_array, inner_radius, inner_angle, outer_radius, outer_angle, full_angle){
              var inner_position = polar_to_cartesian(inner_radius, inner_angle, full_angle);
              var outer_position = polar_to_cartesian(outer_radius, outer_angle, full_angle);
              var curviness = 0.5;
              var inner_control_radius = lerp(inner_radius, outer_radius, curviness);
              var outer_control_radius = lerp(outer_radius, inner_radius, curviness);
              var inner_control = polar_to_cartesian(inner_control_radius, inner_angle, full_angle);
              var outer_control = polar_to_cartesian(outer_control_radius, outer_angle, full_angle);
              string_array.push('<path d="M', inner_position.x,' ',inner_position.y,' C ', inner_control.x ,' ',inner_control.y ,', ', outer_control.x ,' ',outer_control.y ,', ', outer_position.x,' ',outer_position.y,'"/>\n');
            };
            var push_circle = function(string_array, position, radius){ string_array.push('<circle cx="',position.x,'" cy="',position.y,'" r="',radius,'"/>\n'); };
            var push_text = function(string_array, input_string, text_size, attributes, alignment = 0.5) {
              var lines = input_string.split('\n');
              if(!(lines.length > 0)) return;
              var line_height = text_size * 1.5;
              string_array.push('<text ');
              string_array.push.apply(string_array, attributes);
              string_array.push('>');
              var total_height = line_height*(lines.length-1) + text_size;
              var y = lerp(text_size, text_size-total_height, alignment);
              string_array.push('<tspan x="0" y="',y,'">',lines[0],'</tspan>');
              for(var i = 1; i < lines.length; ++i){ string_array.push('<tspan x="0" dy="',line_height,'">',lines[i],'</tspan>'); }
              string_array.push('</text>\n');
            };
            var generate = function(){
              var angle_offset = 0.5 * (1 + category_spacing + sub_category_spacing);
              var accumulator = angle_offset;
              var categories = tree_text.children || [];
              for(var i = 0; i < categories.length; ++i){
                var category = categories[i];
                var sub_categories = category.children || [];
                for(var k = 0; k < sub_categories.length; ++k){
                  var sub_category = sub_categories[k];
                  var entries = sub_category.children || [];
                  for(var m = 0; m < entries.length; ++m){ entries[m].angle = accumulator; accumulator += 1; }
                  accumulator += sub_category_spacing;
                  sub_category.angle = (entries[0].angle + entries[entries.length-1].angle) * 0.5;
                }
                accumulator += category_spacing;
                category.angle = (sub_categories[0].angle + sub_categories[sub_categories.length-1].angle) * 0.5;
              }
              var full_angle = accumulator - angle_offset;
              var result = [];
              result.push('<rect x="-1200" y="-1100" width="2400" height="2200" fill="transparent" />');
              var text_flipping_attributes = 'text-anchor="end" transform="rotate(180 12,0)"';
              result.push('<circle r="',circumcircle_radius,'" style="stroke:',circumcircle_color,';" fill="none"/>\n');
              result.push('<text style="font-size:',title_text_size,'px; font-weight:bold; text-anchor:middle;" y="',title_y,'">',tree_text.name,'</text>');
              result.push('<g style="font-size:',category_text_size,'px;">\n');
              var default_label_radius = 1000;
              var closer_label_radius = 820;
              for(var i = 0; i < categories.length; ++i){
                  var category = categories[i];
                  var current_radius = default_label_radius;
                  if (category.name === "Quantitative \nSustainability" || category.name === "Systems & Nexus \nModeling" || category.name === "Empirical \n& Field-Based") {
                    current_radius = closer_label_radius;
                  }
                  var pos = polar_to_cartesian(current_radius, category.angle, full_angle);
                  var text_anchor = "middle";
                  var angle_in_rads = category.angle * 2 * Math.PI / full_angle;
                  var cos_val = Math.cos(angle_in_rads);
                  if (cos_val > 0.1) { text_anchor = "start"; } else if (cos_val < -0.1) { text_anchor = "end"; }
                  var attributes = ['style="text-anchor:', text_anchor, '; font-weight:bold; fill:', tree_colors[i % tree_colors.length], '" ','transform="translate(', pos.x.toFixed(1), ',', pos.y.toFixed(1), ')"'];
                  push_text(result, category.name, category_text_size, attributes);
              }
              result.push('</g>');
              result.push('<g style="font-size:',entry_text_size,'px;">\n');
              for(var i = 0; i < categories.length; ++i){
                var category = categories[i];
                result.push('<g style="fill:',tree_colors[i],'; stroke:',tree_colors[i],';">\n');
                var sub_categories = category.children || [];
                for(var k = 0; k < sub_categories.length; ++k){
                  var sub_category = sub_categories[k];
                  result.push('<g>\n');
                  var is_flipped = (sub_category.angle > (full_angle / 2));
                  var entries = sub_category.children || [];
                  for(var m = 0; m < entries.length; ++m){
                    var entry = entries[m];
                    var angle_in_degrees = entry.angle * 360.0 / full_angle - 90.0;
                    if(entry.article){ result.push('<a xlink:href="',link_prefix,entry.article,'">\n'); }
                    result.push('<g transform="rotate(',angle_in_degrees,') translate(',tree_radii[2],')">\n','<circle cx="3" r="',small_dot_radius,'"/>\n','<text x="12" y="4" ',(is_flipped? text_flipping_attributes:''),'>',entry.name,'</text>\n','</g>\n');
                    push_radial_path(result, tree_radii[1], sub_category.angle, tree_radii[2], entry.angle, full_angle);
                    if(entry.article){ result.push('</a>\n'); }
                  }
                  push_radial_path(result, tree_radii[0], category.angle, tree_radii[1], sub_category.angle, full_angle);
                  var position = polar_to_cartesian(circumcircle_radius, sub_category.angle, full_angle);
                  push_circle(result, position, big_dot_radius);
                  var text_attributes = ['style="font-size:',subcategory_text_size,'px;" '];
                  if(is_flipped){ text_attributes.push('transform="translate(',position.x-24,',',position.y,')" text-anchor="end"'); }
                  else{ text_attributes.push('transform="translate(',position.x+24,',',position.y,')"'); }
                  var alignment = 0.5 * (1.0 + Math.cos(sub_category.angle * 2*Math.PI / full_angle));
                  push_text(result, sub_category.name, subcategory_text_size, text_attributes, alignment);
                  result.push('</g>\n');
                }
                result.push('</g>\n');
              }
              result.push('</g>\n');
              var zoomTarget = document.getElementById("zoom-target");
              if (zoomTarget) {
                zoomTarget.innerHTML = result.join('');
              } else {
                console.error('Fatal Error: SVG container with id "zoom-target" was not found.');
              }
            };
            generate();

            // HOVER-TO-ZOOM LOGIC
            const zoomContainer = document.querySelector('.svg-zoom-container');
            const svgElement = document.getElementById('chart-svg');
            const zoomTarget = document.getElementById('zoom-target');

            if (zoomContainer && svgElement && zoomTarget) {
                zoomContainer.addEventListener('mouseenter', () => zoomTarget.classList.add('is-zoomed'));
                zoomContainer.addEventListener('mouseleave', () => zoomTarget.classList.remove('is-zoomed'));
                zoomContainer.addEventListener('mousemove', function(event) {
                    const svgRect = svgElement.getBoundingClientRect();
                    const viewBox = svgElement.viewBox.baseVal;
                    if (!viewBox) return;

                    const mouseX = event.clientX - svgRect.left;
                    const mouseY = event.clientY - svgRect.top;

                    const svgX = mouseX * (viewBox.width / svgRect.width);
                    const svgY = mouseY * (viewBox.height / svgRect.height);

                    const originX = svgX - 1200;
                    const originY = svgY - 1100;

                    // *** THIS IS THE FIX ***
                    // Appended 'px' to the values
                    zoomTarget.style.transformOrigin = `${originX}px ${originY}px`;
                });
            }

        } catch (e) {
            console.error("Error initializing chart:", e);
        }
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', init);
    } else {
        init();
    }
})();
</script>
